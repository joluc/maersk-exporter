1. Get the application URL by running these commands:
{{- if contains "NodePort" .Values.service.type }}
  export NODE_PORT=$(kubectl get --namespace {{ .Release.Namespace }} -o jsonpath="{.spec.ports[0].nodePort}" services {{ include "maersk-exporter.fullname" . }})
  export NODE_IP=$(kubectl get nodes --namespace {{ .Release.Namespace }} -o jsonpath="{.items[0].status.addresses[0].address}")
  echo http://$NODE_IP:$NODE_PORT
{{- else if contains "LoadBalancer" .Values.service.type }}
  export SERVICE_IP=$(kubectl get svc --namespace {{ .Release.Namespace }} {{ include "maersk-exporter.fullname" . }} --template "{{"{{ range (index .status.loadBalancer.ingress 0) }}{{.}}{{ end }}"}}")
  echo http://$SERVICE_IP:{{ .Values.service.port }}
{{- else if contains "ClusterIP" .Values.service.type }}
  export POD_NAME=$(kubectl get pods --namespace {{ .Release.Namespace }} -l "app.kubernetes.io/name={{ include "maersk-exporter.name" . }},app.kubernetes.io/instance={{ .Release.Name }}" -o jsonpath="{.items[0].metadata.name}")
  kubectl --namespace {{ .Release.Namespace }} port-forward $POD_NAME 8080:9878
  echo "Visit http://127.0.0.1:8080/metrics"
{{- end }}

2. Check the exporter health:
  curl http://localhost:8080/healthz

3. View logs:
  kubectl logs -f -l app.kubernetes.io/name={{ include "maersk-exporter.name" . }} -n {{ .Release.Namespace }}

{{- if not .Values.credentials.existingSecret }}

⚠️  WARNING: You are using inline credentials!
For production, create a secret and reference it:

  kubectl create secret generic maersk-exporter-credentials \
    --from-literal=maersk-consumer-key='your-key' \
    --from-literal=aisstream-api-key='your-key' \
    -n {{ .Release.Namespace }}

Then upgrade with:
  helm upgrade {{ .Release.Name }} maersk-exporter \
    --set credentials.existingSecret=maersk-exporter-credentials \
    -n {{ .Release.Namespace }}
{{- end }}
